\section{First section}

\begin{namedformula}{Pythagorean identity}
    \label{formula:pythagorean}
    \sin^2\theta+\cos^2\theta=1
\end{namedformula}

\begin{namedformula}{Quadratic formula}
    \label{formula:quadratic}
    x=\frac{-b\pm\sqrt{b^2-4ac}}{2a}
\end{namedformula}

The first result is \formularef{formula:pythagorean},
titled ``\nameref{formula:pythagorean}''.

The second result is \formularef{formula:quadratic}.
