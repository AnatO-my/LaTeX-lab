\OTWorkstream[status={Agreed scope},owner={Software lead (unassigned)}]
 {WS-A}{Departmental records platform}

 outstanding courses, semester/session results and exports. Mobile remains a later option.}
\OTWorkstream[status={Agreed scope},owner={Hardware/firmware leads (unassigned)}]
 {WS-B}{Standalone biometric terminal}
 {A physical fingerprint terminal with local controller, acquisition/matching,
 status indicators, communications, power and enclosure. Future smart-card/NFC
 identification remains an option.}
\OTWorkstream[status={Agreed scope}]{WS-I}{Student Identity Interface}
 {Join the independent workstreams through a stable internal student identity.
 The terminal reports an identity outcome; the records platform authorises access.}

\OTRequirement[workstream=WS-A,status={Agreed scope},owner={Academic data steward},
 acceptance={A synthetic profile retains names, contacts, school email, guardian details,
 current level and admission year across updates.},related={TEST-01}]
 {REQ-A01}{Student and academic records}
 {Separate university-sourced authoritative data from department-owned extensions.
 Derive outstanding courses only where curriculum rules and source information support it.}
\OTRequirement[workstream=WS-B,status={Agreed scope},
 acceptance={The terminal can demonstrate acquisition and identity outcomes against a mock receiver.},
 related={TEST-02}]
 {REQ-B01}{Independent physical terminal}
 {Keep academic records off the biometric device. Isolate biometric credentials;
 fingerprints are not primary student identifiers.}
\OTRequirement[workstream=WS-I,status={Agreed scope},
 acceptance={An identity match without staff permission does not grant access.},
 related={REQ-A01,REQ-B01,TEST-02}]
 {REQ-I01}{Identity is separate from authorisation}
 {The device returns a stable student ID only on a successful match, plus status and
 minimal metadata. Failure, ambiguous matches and communication errors are explicit outcomes.}
